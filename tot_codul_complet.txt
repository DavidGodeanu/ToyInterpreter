package interpreter.controller;

import interpreter.exceptions.ADTException;
import interpreter.exceptions.ExpressionException;
import interpreter.exceptions.StatementException;
import interpreter.model.adt.IStack;
import interpreter.model.PrgState;
import interpreter.model.stmt.IStmt;
import interpreter.repository.IRepository;

public class Controller {
    private IRepository repo;
    private boolean displayFlag;

    public Controller(IRepository repo, boolean displayFlag) {
        this.repo = repo;
        this.displayFlag = displayFlag;
    }

    public PrgState oneStep(PrgState state) throws StatementException, ExpressionException, ADTException {
        IStack<IStmt> stack = state.getExeStack();
        if (stack.isEmpty()) {
            throw new StatementException("Program state stack is empty");
        }
        IStmt crtStmt = stack.pop();
        return crtStmt.execute(state);
    }

    public void allStep() throws StatementException, ExpressionException, ADTException {
        PrgState prg = repo.getCrtPrg();
        if (displayFlag) {
            System.out.println(prg.toString());
        }
        while (!prg.getExeStack().isEmpty()) {
            oneStep(prg);
            if (displayFlag) {
                System.out.println(prg.toString());
            }
        }
    }
}

package interpreter.model.adt;

import interpreter.exceptions.ADTException;

import java.util.HashMap;
import java.util.Map;

public class MyDictionary<K, V> implements IDictionary<K, V> {
    private final Map<K, V> dict = new HashMap<>();

    @Override
    public boolean isDefined(K key) {
        return dict.containsKey(key);
    }

    @Override
    public V lookup(K key) throws ADTException {
        if (!isDefined(key)) {
            throw new ADTException("Key '" + key + "' is not defined in the dictionary.");
        }
        return dict.get(key);
    }

    @Override
    public void put(K key, V value) {
        dict.put(key, value);
    }

    @Override
    public void update(K key, V value) throws ADTException {
        if (!isDefined(key)) {
            throw new ADTException("Cannot update. Key '" + key + "' is not defined.");
        }
        dict.put(key, value);
    }

    @Override
    public void remove(K key) throws ADTException {
        if (!isDefined(key)) {
            throw new ADTException("Cannot remove. Key '" + key + "' is not defined.");
        }
        dict.remove(key);
    }

    @Override
    public Map<K, V> toMap() {
        return Map.copyOf(dict);
    }

    @Override
    public String toString() {
        return dict.toString();
    }
}

package interpreter.model.exp;

import interpreter.exceptions.ADTException;
import interpreter.exceptions.ExpressionException;
import interpreter.model.adt.IDictionary;
import interpreter.model.type.IntType;
import interpreter.model.value.IValue;
import interpreter.model.value.IntValue;

public record ArithExp(ArithOp op, IExp e1, IExp e2) implements IExp {

    @Override
    public IValue eval(IDictionary<String, IValue> tbl)
            throws ExpressionException, ADTException {

        IValue v1 = e1.eval(tbl);

        if (!v1.getType().equals(new IntType())) {
            throw new ExpressionException("First operand is not an integer.");
        }

        IValue v2 = e2.eval(tbl);

        if (!v2.getType().equals(new IntType())) {
            throw new ExpressionException("Second operand is not an integer.");
        }

        IntValue i1 = (IntValue) v1;
        IntValue i2 = (IntValue) v2;

        int n1 = i1.val();
        int n2 = i2.val();

        return switch (op) {
            case ADD -> new IntValue(n1 + n2);

            case SUB -> new IntValue(n1 - n2);

            case MUL -> new IntValue(n1 * n2);

            case DIV -> {
                if (n2 == 0) {
                    throw new ExpressionException("Division by zero.");
                }

                yield new IntValue(n1 / n2);
            }
        };
    }

    @Override
    public String toString() {
        return "(" + e1 + " " + op + " " + e2 + ")";
    }
}

    public IValue eval(IDictionary<String, IValue> tbl) throws ExpressionException, ADTException {

        return switch (op) {
            case AND -> new BoolValue(n1 && n2);
            case OR -> new BoolValue(n1 || n2);
        };
    }

    @Override
    public String toString() {
        return "(" + e1 + " " + op + " " + e2 + ")";
    }

    @Override

            case LT -> n1 < n2;
            case LE -> n1 <= n2;
            case EQ -> n1 == n2;
            case NE -> n1 != n2;
            case GT -> n1 > n2;
            case GE -> n1 >= n2;
        });
    }

    @Override
    public String toString() {
        return "(" + e1 + " " + op + " " + e2 + ")";
    }
}

import interpreter.exceptions.ExpressionException;
import interpreter.exceptions.StatementException;
import interpreter.model.adt.IDictionary;
import interpreter.model.exp.IExp;
import interpreter.model.PrgState;
import interpreter.model.value.IValue;

public record AssignStmt(String id, IExp iExp) implements IStmt {
    @Override
    public PrgState execute(PrgState state) throws StatementException, ExpressionException, ADTException {
        IDictionary<String, IValue> symTable = state.getSymTable();
        if (!symTable.isDefined(id)) {
            throw new StatementException("Variable '" + id + "' is not declared.");
        }
        IValue val = iExp.eval(symTable);
        IValue expectedType = symTable.lookup(id);
        if (!val.getType().equals(expectedType.getType())) {
            throw new StatementException("Type of expression and type of variable '" + id + "' do not match.");
        }
        symTable.update(id, val);
        return state;
    }

    @Override
    public String toString() {
        return id + "=" + iExp.toString();
    }
}

public record IfStmt(IExp cond, IStmt thens, IStmt elses) implements IStmt {
    @Override
    public PrgState execute(PrgState state) throws StatementException, ExpressionException, ADTException {
        IValue val = cond.eval(state.getSymTable());
        if (!val.getType().equals(new BoolType())) {
            throw new StatementException("Condition is not a boolean.");
        }
        if (((BoolValue) val).val()) {
            state.getExeStack().push(thens);
        } else {
            state.getExeStack().push(elses);
        }
        return state;
    }

    @Override

        return "(if(" + cond + ") then(" + thens + ") else(" + elses + "))";

            if (!scanner.hasNextLine()) {
                break;
            }
            String input = scanner.nextLine();
            switch (input) {
                case "1" -> {
